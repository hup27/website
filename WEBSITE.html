    <input type="text" id="destination" placeholder="e.g., Paris, Japan, New York">


    <label for="days">Number of Days:</label>

    <input type="number" id="days" min="1" placeholder="e.g., 5">


    <label for="preferences">Preferences (optional):</label>

    <textarea id="preferences" rows="3" placeholder="e.g., museums, beaches, food"></textarea>


    <button onclick="generatePlan()">Generate Trip Plan</button>


    <div id="aiPlan" class="ai-response" style="display:none;"></div>

  </main>


  <script>

    function generatePlan() {

      const destination = document.getElementById('destination').value.trim();

      const days = document.getElementById('days').value.trim();

      const prefs = document.getElementById('preferences').value.trim();


      if (!destination || !days) {

        alert('Please enter destination and number of days!');

        return;

      }


      const suggestions = [

        "Visit local landmarks and famous attractions",

        "Try the local cuisine",

        "Explore hidden gems off the tourist path",

        "Join a local tour or experience",

        "Relax at a park or beach",

        "Check out local events and festivals"

      ];


      let plan = `Your ${days}-day trip to ${destination}:\n\n`;

      for (let i = 1; i <= days; i++) {

        const activity = suggestions[Math.floor(Math.random() * suggestions.length)];

        plan += `Day ${i}: ${activity}${prefs ? " with focus on " + prefs : ""}.\n`;

      }


      const aiDiv = document.getElementById('aiPlan');

      aiDiv.style.display = 'block';

      aiDiv.innerText = plan;

    }

  </script>
